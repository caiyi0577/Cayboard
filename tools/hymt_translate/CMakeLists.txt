cmake_minimum_required(VERSION 3.22)
project(cyhymt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
set(LLAMA_SOURCE_DIR "" CACHE PATH "Pinned llama.cpp STQ1_0 source")
add_subdirectory(${LLAMA_SOURCE_DIR} llama EXCLUDE_FROM_ALL)
add_library(cyhymt SHARED hymt_jni.cpp)
target_link_libraries(cyhymt PRIVATE llama log)
target_compile_options(cyhymt PRIVATE -O3 -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(cyhymt PRIVATE -Wl,--gc-sections -Wl,--strip-all -Wl,--exclude-libs,ALL)
add_executable(hymt-cli EXCLUDE_FROM_ALL smoke.cpp)
target_link_libraries(hymt-cli PRIVATE llama log)
target_compile_options(hymt-cli PRIVATE -O3 -ffunction-sections -fdata-sections)
target_link_options(hymt-cli PRIVATE -Wl,--gc-sections -Wl,--strip-all)
